#include // Function using Call by Value int increment_by_value(int x) { x = x + 1; // (Call by value: x is local) return x; // Returns new value; original remains unchanged } // Function using Call by Reference void increment_by_reference(int *x) { *x = *x + 1; // (Call by reference: modifies original) } int main() { int a = 5, b = 5; int result1 = increment_by_value(a); increment_by_reference(&b); printf("Call by Value: a = %d, result1 = %d\n", a, result1); printf("Call by Reference: b = %d\n", b); return 0; }
