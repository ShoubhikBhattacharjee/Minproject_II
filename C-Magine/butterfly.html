#include <stdio.h>

      void butterflyPattern(int n) {
          // Upper half of the butterfly
          for (int i = 1; i <= n; i++) {
              // Left side of the butterfly
              for (int j = 1; j <= i; j++) {
                  printf("*");
              }
              // Space between two halves
              for (int j = 1; j <= (2 * n - 2 * i); j++) {
                  printf(" ");
              }
              // Right side of the butterfly
              for (int j = 1; j <= i; j++) {
                  printf("*");
              }
              printf("\\n");
          }

          // Lower half of the butterfly
          for (int i = n; i >= 1; i--) {
              // Left side of the butterfly
              for (int j = 1; j <= i; j++) {
                  printf("*");
              }
              // Space between two halves
              for (int j = 1; j <= (2 * n - 2 * i); j++) {
                  printf(" ");
              }
              // Right side of the butterfly
              for (int j = 1; j <= i; j++) {
                  printf("*");
              }
              printf("\\n");
          }
      }

      int main() {
          int n;
          printf("Enter the number of rows: ");
          scanf("%d", &n);
          butterflyPattern(n);
          return 0;
      }
