Interactive Memory Management Simulation
#include <stdio.h>
#include <stdlib.h>

int main() {
  int *ptr;
  ptr = (int*) malloc(sizeof(int));  // malloc

  *ptr = 100;
  printf("Malloc Value: %d\n", *ptr);

  ptr = (int*) realloc(ptr, 3 * sizeof(int));  // realloc
  ptr[1] = 200;
  ptr[2] = 300;

  printf("Realloc Values: %d %d %d\n", ptr[0], ptr[1], ptr[2]);

  free(ptr);  // free
  return 0;
}
