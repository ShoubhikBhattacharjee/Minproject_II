C File Copy Program Code

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
            
int main() {
    char srcFile[50], destFile[50], ch;
    FILE *fs, *fd;
            
    // Step 1: Get file names from user
    printf("Enter source file name: ");
    scanf("%s", srcFile);
    printf("Enter destination file name: ");
    scanf("%s", destFile);

    // Step 2: Open source file for writing
    fp = fopen(srcFile, "a");
    printf("Type source file content here...");

    fprintf(fp, "%s\n", data);
    fclose(fp);
            
    // Step 3: Open source file for reading
    fs = fopen(srcFile, "r");
    if (fs == NULL) {
        printf("Error opening source file!\n");
        return 1;
    }
            
    // Step 4: Open destination file for writing(copying)
    fd = fopen(destFile, "w");
    if (fd == NULL) {
        printf("Error creating destination file!\n");
        fclose(fs);
        return 1;
    }
            
    // Step 5: Copy contents from source to destination
    while ((ch = fgetc(fs)) != EOF) {
        fputc(ch, fd);
    }
            
    // Step 6: Close both files
    fclose(fs);
    fclose(fd);
            
    printf("File copied successfully!\n");
    return 0;
}
