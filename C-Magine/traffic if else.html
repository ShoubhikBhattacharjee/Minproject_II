int main() {
    int light = 0;

    while(1) {
        if (light == 0) {
            printf("Red Light\n");
            light = 1; // Switch to yellow
        } 
        else if (light == 1) {
            printf("Yellow Light\n");
            light = 2; // Switch to green
        } 
        else if (light == 2) {
            printf("Green Light\n");
            light = 0; // Switch to red
        }
        sleep(5); // Wait for 5 seconds before changing lights (slower)
    }
    return 0;
}
