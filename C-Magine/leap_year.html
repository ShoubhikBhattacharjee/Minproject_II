#include <stdio.h>
int main() {
    int year;
    printf("Enter a year: ");
    scanf("%d", &year);

    if (year % 4 == 0) {
        // Line 5 - Check divisibility by 4
        if (year % 100 != 0 || year % 400 == 0) {
            // Line 6 - Check second condition for leap year
            printf("%d is a Leap Year.\\n", year); // Line 7 - Output leap year
        } else {
            // Line 8 - Not a leap year
            printf("%d is not a Leap Year.\\n", year); // Line 9 - Output non-leap year
        }
    } else {
        // Line 10 - Not divisible by 4 (fails first condition)
        printf("%d is not a Leap Year.\\n", year); // Line 11 - Output non-leap year
    }
    return 0;
}
