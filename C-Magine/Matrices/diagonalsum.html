#include <stdio.h> #include <stdlib.h> int main() { int n; printf("Enter the size of matrix: "); scanf("%d", &n); int matrix[n][n]; for (int i = 0; i < n; i++) { for (int j = 0; j < n; j++) { printf("Enter element at [%d][%d]: ", i, j); scanf("%d", &matrix[i][j]); } } int sum = 0; for (int i = 0; i < n; i++) { sum += matrix[i][i]; } printf("Sum of diagonal elements is: %d\n", sum); return 0; }
