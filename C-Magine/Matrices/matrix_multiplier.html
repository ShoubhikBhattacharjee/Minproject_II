#include <stdio.h> #include <stdlib.h> int main() { int n; printf("Enter the size of matrix: "); scanf("%d", &n); int A[n][n], B[n][n], product[n][n]; printf("Enter elements for Matrix A:\n"); for (int i = 0; i < n; i++) { for (int j = 0; j < n; j++) { scanf("%d", &A[i][j]); } } printf("Enter elements for Matrix B:\n"); for (int i = 0; i < n; i++) { for (int j = 0; j < n; j++) { scanf("%d", &B[i][j]); } } // Multiply matrices A and B for (int i = 0; i < n; i++) { for (int j = 0; j < n; j++) { product[i][j] = 0; for (int k = 0; k < n; k++) { product[i][j] += A[i][k] * B[k][j]; } } } printf("The product matrix is:\n"); for (int i = 0; i < n; i++) { for (int j = 0; j < n; j++) { printf("%d ", product[i][j]); } printf("\n"); } return 0; }
