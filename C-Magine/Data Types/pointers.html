Interactive Pointers & Double Pointers Simulation
#include <stdio.h>
#include <stdlib.h>

int main() {
    int *ptr;      // Declare pointer variable 'ptr'
    int var;       // Declare integer variable 'var'

    // At declaration, both are uninitialized (simulated as NULL)
    printf("Enter value for var: ");
    scanf("%d", &var);    // User inputs value for 'var'

    ptr = &var;           // Assign address of 'var' to pointer 'ptr'

    printf("Before pointer-based update:\n");
    printf("var = %d\n", var);
    printf("*ptr = %d\n", *ptr);

    *ptr = var + 10;      // Update value via pointer

    printf("After pointer-based update:\n");
    printf("var = %d\n", var);
    printf("*ptr = %d\n", *ptr);

    // ----- Double Pointer Extension -----
    int **dptr;          // Declare double pointer variable 'dptr'
    dptr = &ptr;         // Assign address of 'ptr' to double pointer 'dptr'

    printf("Before double pointer update:\n");
    printf("var = %d\n", var);
    printf("*ptr = %d\n", *ptr);
    printf("**dptr = %d\n", **dptr);

    printf("Enter new value for var via double pointer: ");
    int newVal;
    scanf("%d", &newVal);    // User inputs new value for var via double pointer
    **dptr = newVal;         // Update 'var' via double pointer

    printf("After double pointer update:\n");
    printf("var = %d\n", var);
    printf("*ptr = %d\n", *ptr);
    printf("**dptr = %d\n", **dptr);

    return 0;
}
