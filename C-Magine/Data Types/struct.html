#include 
struct Student {
    char name[20];
    int age;
    float grade;
};

int main() {
    struct Student s1 = {"Alice", 20, 9.2};

    printf("Name: %s\\n", s1.name);
    printf("Age: %d\\n", s1.age);
    printf("Grade: %.1f\\n", s1.grade);

    return 0;
}
