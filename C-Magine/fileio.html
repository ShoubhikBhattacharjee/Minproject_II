C Program for File Operations

1: #include 
2: int main() {
3:     FILE *fp;
4:     char filename[50], data[100];
5:
6:     // Step 1: Create a file
7:     printf("Enter file name to create: ");
8:     scanf("%s", filename);
9:     fp = fopen(filename, "a");  // Open file in append mode to write at the end
10:    if (fp == NULL) {
11:        printf("File creation failed!\n");
12:        return 1;
13:    }
14:
15:    // Step 2: Write to file
16:    printf("Enter data to write: ");
17:    scanf(" %[^\n]", data);
18:    fprintf(fp, "%s\n", data);  // Write data with newline
19:    fclose(fp);       // fclose() closes the file safely
20:
21:    // Step 3: Read from file
22:    fp = fopen(filename, "r");
23:    if (fp == NULL) {
24:        printf("File read failed!\n");
25:        return 1;
26:    }
27:    printf("Reading file contents:\n");
28:    while (fgets(data, sizeof(data), fp) != NULL) {
29:        printf("%s", data);
30:    }
31:    fclose(fp);  // fclose() closes the file
32:    return 0;
33: }
