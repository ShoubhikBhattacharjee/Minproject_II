#include <stdio.h>
int main() {
    int n;
    printf("Enter number of rows: ");
    scanf("%d", &n);  // User input for rows
    for (int i = 1; i <= n; i++) {  // Outer loop
        for (int j = 1; j <= i; j++) {  // Inner loop
            printf("* ");
        }
        printf("\n");
    }
    return 0;
}
