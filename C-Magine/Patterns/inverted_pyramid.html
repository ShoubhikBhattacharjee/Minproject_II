Code Visualization

#include <stdio.h>
int main() {
    int n = 5; // Number of rows
    for (int i = n; i >= 1; i--) {  // Outer loop
        for (int j = 1; j <= i; j++) {  // Inner loop
            printf("* ");
        }
        printf("\n");
    }
    return 0;
}
