Variable Scope Simulation
#include <stdio.h>

int globalVar = 10; // Global variable

void demoFunction() {
    int localVar = 5; // Local variable
    static int staticVar = 0; // Static variable
    staticVar++;

    printf("Local: %d, Static: %d, Global: %d\n", localVar, staticVar, globalVar);
}

int main() {
    demoFunction();
    demoFunction();
    return 0;
}
