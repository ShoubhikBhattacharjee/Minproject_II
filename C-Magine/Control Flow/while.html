int main() {
    int progress = 0; // Starting point

    while (progress <= 100) {
        printf("Progress: %d%%\n", progress); // Display progress
        progress += 10; // Increment progress by 10
        sleep(1); // Simulate task delay (1 second)
    }

    printf("Task Complete!\n");
    return 0;
}
